\subsection{定理与证明}

\begin{frame}{定理、证明与例题}
  \begin{theorem}[勾股定理]
    在直角三角形中，两条直角边的平方和等于斜边的平方。
  \end{theorem}
  \begin{proof}
    设两直角边长为 $a,b$，斜边长为 $c$。取四个全等直角三角形拼成边长
    $a+b$ 的正方形，比较两种剖分的面积即得 $a^2+b^2=c^2$。
  \end{proof}
  \begin{example}
    当 $a=3$，$b=4$ 时，$c=\sqrt{a^2+b^2}=5$。
  \end{example}
\end{frame}

\subsection{公式排版}

\begin{frame}{行内与行间公式}
  \small
  行内公式：目标函数记为 $J(\theta)$，学习率取 $\alpha=0.05$，样本量 $n=320$。
  \[ k:[-\pi,\pi]\rightarrow[0,1] \]
  编号公式：网络输入为图像张量
  \begin{equation}
    \mathbf{x}\in\mathbb{R}^{C_{\text{in}}\times H\times W}
    \label{eq:input}
  \end{equation}
  其中 $C_{\text{in}}$ 表示通道数，$H$ 表示图像高度，$W$ 表示图像宽度；
  由式~\eqref{eq:input} 可确定各层张量形状。
\end{frame}

\begin{frame}{多行公式与分段函数}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small 多行对齐（align）：
      \begin{align}
        Q_{\text{target}} &= r+\gamma\,Q^{\pi}(s',\pi_\theta(s'))+\epsilon,\\
        \epsilon &\sim \mathrm{clip}\bigl(\mathcal{N}(0,\sigma),-c,c\bigr).
      \end{align}
    \end{column}
    \begin{column}{0.45\textwidth}
      \small 分段函数（cases）：
      \[ L(e)=\begin{cases}
        e^{2}, & |e|\le 1;\\
        2|e|-1, & |e|>1.
      \end{cases} \]
    \end{column}
  \end{columns}
  \vskip1ex
  {\small 其中 $r$ 为即时奖励，$\gamma$ 为折扣因子，$Q^{\pi}$ 为动作价值函数；
    $L(e)$ 为鲁棒损失，随误差 $e$ 分段变化。}
\end{frame}
