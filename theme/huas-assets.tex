% Internal asset loader; the public command is \HUASsetlogo{<image path>}.
\let\huas@logofile\@empty
\newcommand{\HUASsetlogo}[1]{%
  \IfFileExists{#1}{\def\huas@logofile{#1}}{%
    \let\huas@logofile\@empty
    \PackageWarning{beamerthemeHUAS}{Logo image not found: #1}%
  }%
}
\newcommand{\huas@insertlogo}[1][0.34cm]{%
  \ifx\huas@logofile\@empty\else
    \includegraphics[width=0.15\textwidth,height=#1,keepaspectratio]{\huas@logofile}%
  \fi
}

% User-supplied artwork is optional and keeps its original colors/transparency.
\let\huas@buildingfile\@empty
\newcommand{\HUASsetbuilding}[1]{%
  \if\relax\detokenize{#1}\relax
    \let\huas@buildingfile\@empty
  \else
    \IfFileExists{#1}{\def\huas@buildingfile{#1}}{%
      \let\huas@buildingfile\@empty
      \PackageWarning{beamerthemeHUAS}{Building image not found: #1}%
    }%
  \fi
}
\newcommand{\huas@insertbuilding}{%
  \ifx\huas@buildingfile\@empty
    % No artwork is invented when an asset is absent.
  \else
    \includegraphics[width=\linewidth,height=0.20\paperheight,keepaspectratio]{\huas@buildingfile}%
  \fi
}
