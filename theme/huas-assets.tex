% Internal asset loader; the public command is \HUASsetlogo{<image path>}.
\let\huas@logofile\@empty
\newcommand{\HUASsetlogo}[1]{%
  \IfFileExists{#1}{\def\huas@logofile{#1}}{%
    \let\huas@logofile\@empty
    \PackageWarning{beamerthemeHUAS}{Logo image not found: #1}%
  }%
}
\newcommand{\huas@insertlogo}[1][0.34cm]{%
  \ifx\huas@logofile\@empty\else
    \includegraphics[height=#1]{\huas@logofile}%
  \fi
}
