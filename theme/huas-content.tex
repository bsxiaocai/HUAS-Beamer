% Keep native Beamer environments; only attribution and a complete key-point
% frame need project commands. No verbatim wrapper or bibliography backend.
\setbeamertemplate{blocks}[default]
\setbeamertemplate{caption}[numbered]
\setbeamertemplate{bibliography item}[text]
\setbeamertemplate{section in toc}[sections numbered]
\newenvironment{HUASquotation}[1]{%
  \def\huas@quotesource{#1}%
  % 跟随当前栏宽，故双栏中不会使用整页宽度。
  \begin{beamercolorbox}[wd=\linewidth,sep=\huas@layout{quotation padding}]{HUAS quotation}
    \usebeamerfont{quotation}%
}{%
    \par\medskip
    {\raggedleft\usebeamerfont{HUAS source}\usebeamercolor[fg]{HUAS source}%
      \huas@quotesource\par}%
  \end{beamercolorbox}%
}
\newcommand{\HUASkeypoint}[2]{%
  \begin{frame}{#1}
    \begin{beamercolorbox}[wd=\textwidth,sep=\huas@layout{keypoint padding}]{HUAS quotation}
      \usebeamerfont{HUAS key point}#2\par
    \end{beamercolorbox}
  \end{frame}%
}
